def bubble_sort_with_steps(arr):
    # Создаем копию массива
    arr = arr.copy()
    n = len(arr)
    
    print(f"Исходный массив: {arr}\n")
    
    for i in range(n - 1):
        print(f"Итерация {i + 1}:")
        swapped = False
        
        for j in range(n - 1 - i):
            if arr[j] > arr[j + 1]:
                # Меняем элементы местами
                arr[j], arr[j + 1] = arr[j + 1], arr[j]
                print(f"{arr} ({arr[j + 1]} и {arr[j]} поменялись)")
                swapped = True
            else:
                print(f"{arr} ({arr[j]} и {arr[j + 1]} не меняются)")
        
        if not swapped:
            break
        print()
    
    print(f"\nОтсортированный массив: {arr}")
    return arr


# Исходный массив
arr = [7, 3, 8, 1, 4, 6, 2, 5]
bubble_sort_with_steps(arr)
